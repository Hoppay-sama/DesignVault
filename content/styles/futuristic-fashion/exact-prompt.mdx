```javascript
// Mouse spotlight tracking
let mouseX = 0, mouseY = 0, smoothX = 0, smoothY = 0;
const ease = 0.1;
function updateSpotlight() {
  smoothX += (mouseX - smoothX) * ease;
  smoothY += (mouseY - smoothY) * ease;
  document.documentElement.style.setProperty('--mouse-x', smoothX + 'px');
  document.documentElement.style.setProperty('--mouse-y', smoothY + 'px');
  requestAnimationFrame(updateSpotlight);
}
```
