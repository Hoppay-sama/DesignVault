## SCROLL CHOREOGRAPHY (JS)

```javascript
// Lerp smoothing
let current = 0, target = 0;
const lerpFactor = 0.14;

function segmentInOut(scroll, start, end) {
  const t = (scroll - start) / (end - start);
  if (t < 0 || t > 1) return 0;
  if (t < 0.3) return t / 0.3;
  if (t > 0.7) return (1 - t) / 0.3;
  return 1;
}

function update() {
  current += (target - current) * lerpFactor;
  const scroll = current;

  // Layer 1: Sky — slow drift
  setTransform('sky', { ty: -scroll * 0.3 * 0.08, scale: 1 + scroll * 0.00002 });

  // Layer 2: Back architecture
  setTransform('back-four', { ty: -scroll * 0.5 * 0.12, scale: 1 + scroll * 0.00003 });

  // Layer 3: Bazaar — saturation ramp
  const bazaarSat = 60 + segmentInOut(scroll, 1760, 2700) * 40;
  document.getElementById('bazaar').style.filter = `saturate(${bazaarSat}%)`;

  // Layer 5: Bridge — blur ramp on exit
  const bridgeBlur = segmentInOut(scroll, 1200, 1620) * 14;
  const bridgeBright = 1 - segmentInOut(scroll, 1200, 1620) * 0.15;
  document.getElementById('bridge').style.filter = `blur(${bridgeBlur}px) brightness(${bridgeBright})`;

  // Title — rise and fade
  const titleProgress = segmentInOut(scroll, 0, 650);
  document.getElementById('title').style.transform = `translateY(${-210 * titleProgress}px)`;
  document.getElementById('title').style.opacity = 1 - titleProgress;

  // Story panel — fade in
  const storyProgress = segmentInOut(scroll, 1760, 2700);
  document.getElementById('story-1').style.opacity = storyProgress;
  document.getElementById('story-1').style.transform = `translateY(${40 - storyProgress * 40}px)`;

  // Sights slider — fly in from 420vw
  const sliderProgress = segmentInOut(scroll, 2760, 3560);
  const sliderX = 420 - Math.pow(sliderProgress, 1.55) * 420;
  document.getElementById('sights').style.transform = `translateX(${sliderX}vw)`;

  // Nav buttons — fade in
  const navProgress = Math.max(0, Math.min(1, (scroll - 3360) / 140));
  document.getElementById('nav').style.opacity = navProgress;

  requestAnimationFrame(update);
}

window.addEventListener('scroll', () => { target = window.scrollY; });
update();
```

- **2760-3560px:** Sights slider flies in from 420vw with power(1.55) easing. Cards show city landmarks.
